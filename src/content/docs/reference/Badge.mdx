import { Badge as ReactBadge } from '@/components/ui/Badge'
import AstroBadge from '@/components/ui/Badge.astro'
import SvelteBadge from '@/components/ui/Badge.svelte'

A badge component for displaying status, labels, and other small pieces of information.

## Usage

### Astro

<div class="space-y-4">
  <div class="flex items-center gap-2">
    <span>Status:</span>
    <AstroBadge variant="secondary">Active</AstroBadge>
  </div>
  
  <div class="flex items-center gap-2">
    <span>Priority:</span>
    <AstroBadge variant="destructive">High</AstroBadge>
  </div>
  
  <div class="flex items-center gap-2">
    <span>Category:</span>
    <AstroBadge variant="outline">Design</AstroBadge>
  </div>
</div>

<div class="space-y-4">
