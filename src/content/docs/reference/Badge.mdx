import { Badge as ReactBadge } from '@/components/ui/Badge'
import AstroBadge from '@/components/ui/Badge.astro'
import SvelteBadge from '@/components/ui/Badge.svelte'

A badge component for displaying status, labels, and other small pieces of information.

## Usage

### Astro

<div class="flex flex-wrap gap-2">
  <AstroBadge>Default</AstroBadge>
  <AstroBadge variant="secondary">Secondary</AstroBadge>
  <AstroBadge variant="destructive">Destructive</AstroBadge>
  <AstroBadge variant="outline">Outline</AstroBadge>
</div>

### React

<div class="flex flex-wrap gap-2">
  <ReactBadge>Default</ReactBadge>
  <ReactBadge variant="secondary">Secondary</ReactBadge>
  <ReactBadge variant="destructive">Destructive</ReactBadge>
  <ReactBadge variant="outline">Outline</ReactBadge>
</div>

### Svelte

<div class="flex flex-wrap gap-2">
  <SvelteBadge>Default</SvelteBadge>
  <SvelteBadge variant="secondary">Secondary</SvelteBadge>
  <SvelteBadge variant="destructive">Destructive</SvelteBadge>
  <SvelteBadge variant="outline">Outline</SvelteBadge>
</div>
